    <script>
      // Cálculo automático de la cuota según fórmula financiera
      function calcularCuotaSugerida() {
        const monto = parseFloat(document.getElementById("monto_otorgado").value);
        const tasaAnual = parseFloat(document.getElementById("tasa_interes").value);
        const plazo = parseInt(document.getElementById("plazo_meses").value, 10);

        if (monto > 0 && tasaAnual > 0 && plazo > 0) {
          const iMensual = (tasaAnual / 100) / 12;
          const cuota = monto * (iMensual / (1 - Math.pow(1 + iMensual, -plazo)));
          document.getElementById("cuota_programada").value = cuota.toFixed(2);
        }
      }

      // Sugerir estado del crédito automáticamente según días de atraso
      function actualizarEstadoAutomatico() {
        const dias = parseInt(document.getElementById("dias_atraso").value, 10) || 0;
        const selector = document.getElementById("estado_credito");
        if (dias === 0) {
          selector.value = "Al día";
        } else if (dias > 90) {
          selector.value = "Vencido";
        } else {
          selector.value = "Atrasado";
        }
      }

      function enviarFormulario(e) {
        e.preventDefault();
        const btn = document.getElementById("btnGuardar");
        btn.disabled = true;
        btn.innerText = "Guardando en base de datos...";

        const datos = {
          departamento: document.getElementById("departamento").value,
          edad_cliente: document.getElementById("edad_cliente").value,
          tipo_empleo: document.getElementById("tipo_empleo").value,
          ingreso_mensual: document.getElementById("ingreso_mensual").value,
          monto_otorgado: document.getElementById("monto_otorgado").value,
          tasa_interes: document.getElementById("tasa_interes").value,
          plazo_meses: document.getElementById("plazo_meses").value,
          cuota_programada: document.getElementById("cuota_programada").value,
          monto_pagado: document.getElementById("monto_pagado").value,
          dias_atraso: document.getElementById("dias_atraso").value,
          estado_credito: document.getElementById("estado_credito").value,
          fecha_desembolso: document.getElementById("fecha_desembolso").value,
          fecha_ultimo_pago: document.getElementById("fecha_ultimo_pago").value
        };

        google.script.run
          .withSuccessHandler(function(resp) {
            btn.disabled = false;
            btn.innerText = "Guardar Operación de Crédito";
            const resDiv = document.getElementById("resultado");
            if (resp.exito) {
              resDiv.className = "alert alert-success mt-3 text-center";
              resDiv.innerText = resp.mensaje;
              document.getElementById("creditForm").reset();
            } else {
              resDiv.className = "alert alert-danger mt-3 text-center";
              resDiv.innerText = resp.mensaje;
            }
          })
          .registrarCredito(datos);
      }
    </script>
